NODE.JS EXAMPLE 
const res = await fetch('https://api.jhx.my.id/gd-downloader', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ url: 'https://drive.google.com/file/d/…/view' })
});
const json = await res.json();
if (json.Status) {
  console.log(json.data.file_name, json.data.file_size);
  console.log(json.data.download_url);
}
