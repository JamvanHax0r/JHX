  -d '{"url":"https://www.mediafire.com/file/xxxxx/file.zip"}'

NODE.JS EXAMPLE 
const res = await fetch('https://api.jhx.my.id/mf-downloader', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ url: 'https://www.mediafire.com/file/xxxxx/file.zip' })
});
const json = await res.json();
if (json.Status) {
  console.log(json.data.file_name, json.data.file_size);
  console.log(json.data.url); // proxy link — langsung unduh
}

catatan
  • → Tanpa login MediaFire, tanpa lewati iklan/redirect.
  • → Filename + ukuran auto-terbaca dari halaman file.
  • → Link unduhan via proxy JH — aktif ±2 jam & support resume.
  • 🔥 Direct link MediaFire tidak bocor ke user — aman & privat.
