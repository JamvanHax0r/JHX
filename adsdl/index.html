NODE.JS EXAMPLE 
const res = await fetch('https://api.jhx.my.id/ads-skipper', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ url: 'https://work.ink/...' })
});
const json = await res.json();
if (json.Status) {
  console.log('Bypassed:', json.data.bypassed_url);
}
