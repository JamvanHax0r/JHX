NODE.JS EXAMPLE 
const b64 = require('fs').readFileSync('foto.jpg').toString('base64');

const res = await fetch('https://api.jhx.my.id/hd-image', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ b64, ct: 'image/jpeg' })
});
const json = await res.json();
if (json.Status) console.log(json.data.url);
