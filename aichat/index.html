NODE.JS EXAMPLE 
const res = await fetch('https://api.jhx.my.id/ai-chat', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ message: 'Review kode ini dong', files: [{ name: 'app.js', content: '...' }] })
});
const json = await res.json();
if (json.Status) console.log(json.data.reply);
